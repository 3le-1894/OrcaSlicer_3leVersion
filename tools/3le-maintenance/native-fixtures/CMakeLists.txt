cmake_minimum_required(VERSION 3.20)
project(startup_readiness_contract LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
foreach(name page_readiness registration)
 add_executable(${name} ${name}.cpp)
 target_compile_options(${name} PRIVATE -UNDEBUG -include "${CMAKE_CURRENT_SOURCE_DIR}/assertions.hpp")
 add_test(NAME ${name} COMMAND ${name})
endforeach()
